    public int getColorForState(int[] set, int fallback) {
        for (int i = 0; i < states.length; i++) {
            boolean match = true;
            for (int want : states[i]) {
                boolean has = false;
                for (int got : set) if (got == Math.abs(want)) has = true;
                if (want > 0 ? !has : has) match = false;
            }
            if (match) return colors[i];
        }
        return fallback;
    }

    private static int px(String v, int fallback) {
        if (v == null || v.isEmpty()) return fallback;
        if (v.equals("match_parent")) return -1; if (v.equals("wrap_content")) return -2;
        if (v.endsWith("dp")) return Math.round(Float.parseFloat(v.substring(0, v.length() - 2)) * 2.75f);

    }
    private static View build(Context c, Element e, ViewGroup parent, Set<Integer> seen) throws Exception {

        else if (tag.equals("View")) cls = View.class;
        else cls = Class.forName("android.widget." + tag);
        View v = (View) cls.getConstructor(Context.class).newInstance(c);
        String id = e.getAttributeNS(NS, "id");
        if (!id.isEmpty()) {
            String name = id.substring(id.indexOf('/') + 1);
            int value = Class.forName(c.getPackageName() + ".R$id").getField(name).getInt(null);
            if (!seen.add(value)) throw new RuntimeException("duplicate id " + name);

        }

        String vis = e.getAttributeNS(NS, "visibility");
        if (vis.equals("gone")) v.setVisibility(View.GONE); else if (vis.equals("invisible")) v.setVisibility(View.INVISIBLE);
        String alpha = e.getAttributeNS(NS, "alpha"); if (!alpha.isEmpty()) v.setAlpha(Float.parseFloat(alpha));
        if (v instanceof android.widget.TextView) ((android.widget.TextView) v).setText(e.getAttributeNS(NS, "text"));
        if (v instanceof android.widget.LinearLayout) ((android.widget.LinearLayout) v).setOrientation("vertical".equals(e.getAttributeNS(NS, "orientation")) ? 1 : 0);

public class Surface {
    public static boolean simNoHardwareCanvas; public static long simClockOffsetUs; public static int simLocks;
    private final android.media.MediaCodec codec; private Canvas locked; private boolean released;
    public Surface(android.media.MediaCodec codec) { this.codec = codec; }
    public Canvas lockHardwareCanvas() {
        if (released) throw new IllegalStateException("released");
        if (simNoHardwareCanvas) throw new IllegalStateException("Surface was already locked");
        if (locked != null) throw new IllegalStateException("already locked");
        simLocks++;
        locked = new Canvas();
        return locked;
    }
    public void unlockCanvasAndPost(Canvas canvas) {
        if (canvas != locked) throw new IllegalArgumentException("canvas object must be the same instance that was previously returned by lockCanvas");
        if (canvas.saveDepth != 0) throw new IllegalStateException("unbalanced save/restore");
        locked = null;
        codec.simSurfaceFrame(System.nanoTime() / 1000L + simClockOffsetUs);
    }
    public boolean isValid() { return !released; }
    public void release() { released = true; }
}

    private static void findDrawables(File folder, int depth, List<File> out) {

    private static long whole(Object value) {

    static String checkTiming(Result result, Object stamps, long started, long uploaded) {
        List<Mark> marks = result.marks;
        if (marks.isEmpty() || stampCount(stamps) != marks.size()) return "the run's timing marks didn't match the replay";
        long previous = started;
        long pausedMs = 0L;
        long pausedAt = -1L;
        boolean playing = true;
        double slowest = 0.0;
        for (int i = 0; i < marks.size(); i++) {
            Mark mark = marks.get(i);
            Map<String, Object> one = stamp(stamps, i);
            if (one == null) return "the run's timing marks didn't match the replay";
            long time = whole(one.get("t"));
            if (whole(one.get("m")) != mark.kind || !mark.code.equals(one.get("c")) || time == Long.MIN_VALUE) {
                return "the run's timing marks didn't match the replay";
            }
            if (time < previous && i > 0) return "the run's timing marks were out of order";
            if (playing && time - previous > MAX_GAP_MS) {
                return "the run's timing could not be confirmed (no mark for " + (time - previous) / 1000 + " s)";
            }
            if (mark.kind == GameView.MARK_PAUSE) {
                pausedAt = time;
                playing = false;
            } else if (mark.kind == GameView.MARK_RESUME) {
                pausedMs += time - pausedAt;
                playing = true;
            }
            double real = (time - started - pausedMs) / 1000.0;
            if (mark.seconds > real + FAST_SLACK_SEC) return "the run was played faster than real time";
            double lag = real - mark.seconds * SLOW_SHARE;
            if (lag > slowest + SLOW_SLACK_SEC) return "the run was played in slow motion";
            slowest = Math.min(slowest, lag);
            previous = time;
        }
        if (marks.get(marks.size() - 1).kind != GameView.MARK_END) return "the replay has no end mark";
        if (uploaded < previous) return "the run's timing marks were out of order";
        return null;
    }

    public static void main(String[] args) throws Exception {
        String account = System.getenv("FIREBASE_SERVICE_ACCOUNT");
        if (args.length > 0) account = new String(Files.readAllBytes(Paths.get(args[0])), StandardCharsets.UTF_8);
        if (account == null || account.trim().isEmpty()) {
            System.out.println("The FIREBASE_SERVICE_ACCOUNT secret is missing. See LEADERBOARD_SETUP.md.");
            System.exit(1);
        }
        String url = System.getenv("DATABASE_URL");
        if (url == null || url.trim().isEmpty()) url = SettingsActivity.Leaderboard.DATABASE_URL;
        database = url.trim();
        while (database.endsWith("/")) database = database.substring(0, database.length() - 1);
        Map<String, Object> key;
        try {
            key = Json.object(Json.parse(account));
        } catch (IllegalArgumentException e) {
            key = null;
        }
        if (key == null || Json.string(key, "client_email").isEmpty() || Json.string(key, "private_key").isEmpty()) {
            System.out.println("FIREBASE_SERVICE_ACCOUNT is not a service account key file.");
            System.exit(1);
        }
        accessToken = signIn(key);
        long deadline = System.currentTimeMillis() + TIME_BUDGET_MS;
        int code = run(deadline);
        try {
            maintain(System.currentTimeMillis(), deadline + 60000L, false);
            refreshCounts(System.currentTimeMillis());
        } catch (IOException | RuntimeException e) {
            report.add("Cleanup will try again next time: " + e.getMessage());
        }
        for (String line : report) System.out.println(line);
        System.out.println("Verified " + verified + ", rejected " + rejected + ", skipped " + skipped + ", still waiting " + waiting + ".");
        System.out.flush();
        System.exit(code);
    }

    static int run(long deadline) throws IOException {
        Map<String, Object> pending = Json.object(Json.parse(call("GET", "/replays.json?shallow=true", null, false, null).body));
        if (pending == null || pending.isEmpty()) {
            report.add("No runs are waiting.");
            pending = new LinkedHashMap<>();
        }
        for (String user : pending.keySet()) {
            if (System.currentTimeMillis() > deadline) {
                waiting++;
                continue;
            }
            Map<String, Object> slots = Json.object(Json.parse(call("GET", "/replays/" + enc(user) + ".json?shallow=true", null, false, null).body));
            if (slots == null) continue;
            List<String> order = new ArrayList<>(slots.keySet());
            order.sort((x, y) -> {
                int a = x.equals("a") ? 0 : x.equals("w") ? 1 : 2;
                int b = y.equals("a") ? 0 : y.equals("w") ? 1 : 2;
                return a != b ? Integer.compare(a, b) : x.compareTo(y);
            });
            for (String slot : order) {
                try {
                    check(user, slot);
                } catch (IOException | RuntimeException e) {
                    waiting++;
                    report.add("Could not check a run from " + shortId(user) + ": " + e.getMessage());
                }
            }
        }
        try {
            processPurchases(deadline);
        } catch (IOException | RuntimeException e) {
            report.add("Purchases will be tried again next time: " + e.getMessage());
        }
        return 0;
    }

    static String shortId(String user) {
        return user.length() <= 6 ? user : user.substring(0, 6) + "...";
    }

    static String enc(String part) {
        try {
            return URLEncoder.encode(part, "UTF-8").replace("+", "%20");
        } catch (IOException e) {
            throw new IllegalStateException(e);
        }
    }

    static boolean userKey(String key) {
        return key != null && key.matches("[A-Za-z0-9_-]{1,128}");
    }

    static boolean safeKey(String key) {
        return key != null && key.matches("^-?[0-9]{1,20}$");
    }

    static void check(String user, String slot) throws IOException {
        String place = "/replays/" + enc(user) + "/" + enc(slot);
        long rules = Json.number(Json.object(Json.parse("{\"rv\":" + orNull(call("GET", place + "/rv.json", null, false, null).body) + "}")), "rv");
        if (rules > GameView.REPLAY_RULES) {
            waiting++;
            report.add("A run from " + shortId(user) + " needs a newer checker. Copy the latest MainActivity.java into the repository.");
            return;
        }
        Reply reply = call("GET", place + ".json", null, true, null);
        Map<String, Object> entry = Json.object(Json.parse(reply.body));
        if (entry == null) return;
        int score = (int) Json.number(entry, "s");
        boolean classic = Boolean.TRUE.equals(entry.get("c"));
        String seed = Json.string(entry, "r");
        long started = Json.number(entry, "st");
        long sent = Json.number(entry, "t");
        long week = Json.number(entry, "w");
        String data = Json.string(entry, "d");
        int outcome;
        String why;
        String owner = "";
        Replayer.Result result = null;
        boolean board = slot.equals("a") || slot.equals("w");
        boolean challenge = slot.matches("x[0-7]");
        if (!(board || challenge) || score < 1 || score > SettingsActivity.Leaderboard.MAX_SCORE || !safeKey(seed)
                || started <= 0 || sent <= 0 || week == Long.MIN_VALUE || data.isEmpty()) {
            outcome = Replayer.REJECTED;
            why = "the upload was incomplete";
        } else if (exists("/banned/" + enc(user) + ".json")) {
            outcome = Replayer.REJECTED;
            why = BANNED_WHY;
        } else if (!(owner = seedOwner(seed)).isEmpty()) {
            if (owner.equals(user)) {
                if (challenge) awardFromSeed(user, seed, sent, score);
                call("DELETE", place + ".json", null, false, reply.etag);
                return;
            }
            outcome = Replayer.REJECTED;
            why = "this run was already used by another player";
        } else {
            byte[] bytes;
            try {
                bytes = Base64.getDecoder().decode(data);
            } catch (IllegalArgumentException e) {
                bytes = null;
            }
            if (bytes == null) {
                outcome = Replayer.REJECTED;
                why = "the replay could not be read";
            } else {
                result = Replayer.run(bytes, score, classic, challenge);
                outcome = result.verdict;
                why = result.why;
                if (outcome == Replayer.OK) {
                    String timing = Replayer.checkTiming(result, entry.get("h"), started, sent);
                    if (timing != null) {
                        outcome = Replayer.REJECTED;
                        why = timing;
                    }
                }
                if (outcome == Replayer.OK && result.seed != Long.parseLong(seed)) {
                    outcome = Replayer.REJECTED;
                    why = "the replay belongs to a different round";
                }
            }
        }
        Map<String, Object> update = new LinkedHashMap<>();
        Map<String, Object> verdict = new LinkedHashMap<>();
        verdict.put("s", (double) Math.max(0, score));
        verdict.put("c", classic);
        verdict.put("ok", outcome == Replayer.OK);
        verdict.put("why", why);
        Map<String, Object> now = new LinkedHashMap<>();
        now.put(".sv", "timestamp");
        verdict.put("t", now);
        if (board) update.put("verdicts/" + user, verdict);
        Map<String, Object> claim = null;
        if (outcome != Replayer.SKIPPED && safeKey(seed) && owner.isEmpty()) {
            claim = new LinkedHashMap<>();
            claim.put("u", user);
            Map<String, Object> stamp = new LinkedHashMap<>();
            stamp.put(".sv", "timestamp");
            claim.put("t", stamp);
            update.put("seeds/" + seed, claim);
        }
        String name = "";
        int gained = -1;
        if (outcome == Replayer.OK && challenge) {
            gained = awardRound(update, user, sent, score, result.seconds, result.flaps, result.twist, result.restMode,
                    weekBest(user, sent));
            if (claim != null) claim.put("q", true);
        } else if (outcome == Replayer.OK) {
            Map<String, Object> scores = Json.object(Json.parse(call("GET", "/scores/" + enc(user) + ".json", null, false, null).body));
            name = Json.string(scores, "n");
            String key = SettingsActivity.Leaderboard.nameKey(name);
            String holder = key.isEmpty() ? "" : String.valueOf(Json.parse(orNull(call("GET", "/names/" + enc(key) + ".json", null, false, null).body)));
            if (name.isEmpty() || !user.equals(holder)) {
                outcome = Replayer.SKIPPED;
                why = "the leaderboard name could not be confirmed. Set your name again and play another round";
                verdict.put("ok", false);
                verdict.put("why", why);
                update.remove("seeds/" + seed);
            } else if (!SettingsActivity.Leaderboard.nameAllowed(name)) {
                outcome = Replayer.SKIPPED;
                why = "the leaderboard name isn't allowed. Choose another name and play another round";
                verdict.put("ok", false);
                verdict.put("why", why);
                update.remove("seeds/" + seed);
            } else {
                String version = Json.string(entry, "vn");
                if (!version.matches("[0-9][0-9A-Za-z._-]{0,15}")) version = "";
                if (claim != null) {
                    Map<String, Object> facts = new LinkedHashMap<>();
                    facts.put("s", (double) score);
                    facts.put("sec", result.seconds);
                    facts.put("fl", (double) result.flaps);
                    facts.put("md", (double) result.restMode);
                    facts.put("wb", (double) weekBest(user, sent));
                    claim.put("f", facts);
                }
                publish(update, user, name, score, classic, sent, (int) week, result, scores, version);
            }
        }
        if (outcome == Replayer.REJECTED && !BANNED_WHY.equals(why)) {
            if (board) lowerClaim(update, user, score, week);
            strike(update, user, why, sent > 0 ? sent : System.currentTimeMillis());
        }
        if (challenge && outcome != Replayer.OK) news(update, user, 0, 0, false, why);
        call("PATCH", "/.json", Json.write(update), false, null);
        Reply removed = call("DELETE", place + ".json", null, false, reply.etag);
        if (removed.code == 412) report.add("A run from " + shortId(user) + " changed while it was being checked; it will be checked again.");
        String who = name.isEmpty() ? shortId(user) : name;
        if (outcome == Replayer.OK && challenge) {
            verified++;
            report.add("Checked a challenge run of " + score + " by " + who + (gained > 0 ? ": earned " + gained + " coins" : ": no new progress"));
        } else if (outcome == Replayer.OK) {
            verified++;
            report.add("Verified " + score + (classic ? " (Classic)" : "") + " by " + who + String.format(" in %.1f s of play", result.seconds));
        } else if (outcome == Replayer.SKIPPED) {
            skipped++;
            report.add("Skipped " + score + " by " + who + ": " + why);
        } else {
            rejected++;
            report.add("Rejected " + score + " by " + who + ": " + why);
        }
    }

    static void lowerClaim(Map<String, Object> update, String user, int score, long week) throws IOException {
        Map<String, Object> claimed = getMap("/scores/" + enc(user) + ".json");
        if (Json.number(claimed, "s") == score && Json.number(claimed, "vs") != score) {
            lowerAllTime(update, user, getMap("/vscores/" + enc(user) + ".json"));
        }
        Map<String, Object> weekly = getMap("/weekly/" + enc(user) + ".json");
        if (week != Long.MIN_VALUE && Json.number(weekly, "k") == week * SettingsActivity.Leaderboard.WEEK_KEY + score && Json.number(weekly, "vs") != score) {
            lowerWeekly(update, user, weekly, getMap("/vweekly/" + enc(user) + ".json"));
        }
    }

    static long lowerAllTime(Map<String, Object> update, String user, Map<String, Object> verified) {
        long best = Math.max(0L, Json.number(verified, "s"));
        update.put("scores/" + user + "/s", (double) best);
        update.put("scores/" + user + "/vs", best > 0 ? (Object) (double) best : null);
        if (best > 0) {
            update.put("scores/" + user + "/c", Boolean.TRUE.equals(verified.get("c")));
            long seconds = Json.number(verified, "d");
            if (seconds >= 0) update.put("scores/" + user + "/d", (double) seconds);
        }
        return best;
    }

    static long lowerWeekly(Map<String, Object> update, String user, Map<String, Object> claimed, Map<String, Object> verified) throws IOException {
        long week = Json.number(claimed, "w");
        long best = Json.number(verified, "w") == week ? Math.max(0L, Json.number(verified, "s")) : 0L;
        if (best <= 0) {
            remove(update, "weekly/" + user);
            return 0L;
        }
        update.put("weekly/" + user + "/s", (double) best);
        update.put("weekly/" + user + "/k", (double) (week * SettingsActivity.Leaderboard.WEEK_KEY + best));
        update.put("weekly/" + user + "/vs", (double) best);
        update.put("weekly/" + user + "/c", Boolean.TRUE.equals(verified.get("c")));
        long seconds = Json.number(verified, "d");
        if (seconds >= 0) update.put("weekly/" + user + "/d", (double) seconds);
        return best;
    }

    static void strike(Map<String, Object> update, String user, String why, long now) throws IOException {
        Map<String, Object> record = getMap("/meta/strikes/" + enc(user) + ".json");
        long first = Json.number(record, "t");
        long count = Json.number(record, "n");
        if (first == Long.MIN_VALUE || count < 0 || now - first > STRIKE_WINDOW_MS || now < first) {
            first = now;
            count = 0;
        }
        count++;
        if (count >= STRIKES_TO_BAN) {
            Map<String, Object> ban = new LinkedHashMap<>();
            ban.put("auto", true);
            ban.put("t", (double) now);
            ban.put("why", why == null ? "" : why.length() > 200 ? why.substring(0, 200) : why);
            update.put("banned/" + user, ban);
            update.put("meta/strikes/" + user, null);
            report.add("Banned " + shortId(user) + " for " + (AUTO_BAN_MS / DAY_MS) + " days after " + count + " rejected runs in a week.");
        } else {
            Map<String, Object> next = new LinkedHashMap<>();
            next.put("n", (double) count);
            next.put("t", (double) first);
            update.put("meta/strikes/" + user, next);
        }
    }

    static int lowerStaleClaims(Map<String, Object> scores, Map<String, Object> vscores, Map<String, Object> weekly,
            Map<String, Object> vweekly, long now, long from, Map<String, Object> update) throws IOException {
        Map<String, Object> waitingRuns = getMap("/replays.json?shallow=true");
        int lowered = 0;
        for (String user : new ArrayList<>(scores.keySet())) {
            Map<String, Object> entry = Json.object(scores.get(user));
            if (!userKey(user) || entry == null || waitingRuns.containsKey(user)) continue;
            long score = Json.number(entry, "s");
            long sent = Json.number(entry, "t");
            if (score <= 0 || Json.number(entry, "vs") == score || sent < from || now - sent < CLAIM_GRACE_MS) continue;
            Map<String, Object> verified = Json.object(vscores.get(user));
            if (score <= Math.max(0L, Json.number(verified, "s"))) continue;
            long best = lowerAllTime(update, user, verified == null ? new LinkedHashMap<String, Object>() : verified);
            entry.put("s", (double) best);
            lowered++;
            if (update.size() >= BATCH) flush(update);
        }
        for (String user : new ArrayList<>(weekly.keySet())) {
            Map<String, Object> entry = Json.object(weekly.get(user));
            if (!userKey(user) || entry == null || waitingRuns.containsKey(user)) continue;
            long score = Json.number(entry, "s");
            long sent = Json.number(entry, "t");
            if (score <= 0 || Json.number(entry, "vs") == score || sent < from || now - sent < CLAIM_GRACE_MS) continue;
            Map<String, Object> verified = Json.object(vweekly.get(user));
            long week = Json.number(entry, "w");
            if (verified != null && Json.number(verified, "w") == week && score <= Json.number(verified, "s")) continue;
            long best = lowerWeekly(update, user, entry, verified == null ? new LinkedHashMap<String, Object>() : verified);
            if (best <= 0) {
                weekly.remove(user);
            } else {
                entry.put("s", (double) best);
                entry.put("k", (double) (week * SettingsActivity.Leaderboard.WEEK_KEY + best));
            }
            lowered++;
            if (update.size() >= BATCH) flush(update);
        }
        return lowered;
    }

    static int expireBans(long now, Map<String, Object> update) throws IOException {
        int lifted = 0;
        Map<String, Object> bans = getMap("/banned.json");
        for (Map.Entry<String, Object> ban : bans.entrySet()) {
            Map<String, Object> details = Json.object(ban.getValue());
            if (details == null || !Boolean.TRUE.equals(details.get("auto"))) continue;
            long since = Json.number(details, "t");
            if (since == Long.MIN_VALUE || now - since >= AUTO_BAN_MS) {
                remove(update, "banned/" + ban.getKey());
                lifted++;
            }
        }
        Map<String, Object> strikes = getMap("/meta/strikes.json");
        for (Map.Entry<String, Object> record : strikes.entrySet()) {
            long first = Json.number(Json.object(record.getValue()), "t");
            if (first == Long.MIN_VALUE || now - first > STRIKE_WINDOW_MS) remove(update, "meta/strikes/" + record.getKey());
        }
        return lifted;
    }

    static Map<String, Object> getMap(String path) throws IOException {
        Map<String, Object> value = Json.object(Json.parse(orNull(call("GET", path, null, false, null).body)));
        return value == null ? new LinkedHashMap<>() : value;
    }

    static void flush(Map<String, Object> update) throws IOException {
        if (update.isEmpty()) return;
        call("PATCH", "/.json", Json.write(update), false, null);
        update.clear();
    }

    static void remove(Map<String, Object> update, String path) throws IOException {
        update.put(path, null);
        if (update.size() >= BATCH) flush(update);
    }

    static List<String> top(Map<String, Object> branch, int count) {
        List<Map.Entry<String, Object>> entries = new ArrayList<>(branch.entrySet());
        entries.removeIf(e -> !(e.getValue() instanceof Map) || Json.number(Json.object(e.getValue()), "s") <= 0);
        entries.sort((a, b) -> {
            Map<String, Object> x = Json.object(a.getValue()), y = Json.object(b.getValue());
            int byScore = Long.compare(Json.number(y, "s"), Json.number(x, "s"));
            return byScore != 0 ? byScore : Long.compare(Json.number(x, "t"), Json.number(y, "t"));
        });
        List<String> out = new ArrayList<>();
        for (int i = 0; i < entries.size() && i < count; i++) out.add(entries.get(i).getKey());
        return out;
    }

    static Map<String, Object> scoresOf(Map<String, Object> branch) {
        Map<String, Object> out = new LinkedHashMap<>();
        for (Map.Entry<String, Object> e : branch.entrySet()) {
            long score = Json.number(Json.object(e.getValue()), "s");
            if (score > 0 && score <= SettingsActivity.Leaderboard.MAX_SCORE) out.put(e.getKey(), (double) score);
        }
        return out;
    }

    static long newest(Map<String, Object> branch) {
        long newest = 0L;
        for (Object raw : branch.values()) newest = Math.max(newest, Json.number(Json.object(raw), "t"));
        return newest;
    }

    static void refreshCounts(long now) throws IOException {
        Map<String, Object> upTo = getMap("/meta/countedUpTo.json");
        if (upTo.isEmpty()) {
            verifiedChanges.clear();
            return;
        }
        int thisWeek = SettingsActivity.Leaderboard.week(now);
        Map<String, Map<String, Long>> changes = new LinkedHashMap<>();
        Map<String, Object> reached = new LinkedHashMap<>();
        for (String branch : new String[] {"scores", "weekly"}) {
            long since = Math.max(0L, Json.number(upTo, branch));
            Map<String, Object> changed = getMap("/" + branch + ".json?orderBy=%22t%22&startAt=" + since);
            long newest = since;
            Map<String, Long> scores = new LinkedHashMap<>();
            for (Map.Entry<String, Object> e : changed.entrySet()) {
                Map<String, Object> entry = Json.object(e.getValue());
                newest = Math.max(newest, Json.number(entry, "t"));
                long score = Json.number(entry, "s");
                if (score <= 0 || score > SettingsActivity.Leaderboard.MAX_SCORE) continue;
                if (branch.equals("weekly") && Json.number(entry, "w") != thisWeek) continue;
                scores.put(e.getKey(), score);
            }
            changes.put(branch, scores);
            reached.put(branch, (double) newest);
        }
        for (Object[] change : verifiedChanges) {
            String branch = (String) change[0];
            if (branch.equals("vweekly") && (Integer) change[1] != thisWeek) continue;
            changes.computeIfAbsent(branch, b -> new LinkedHashMap<>()).put((String) change[2], (Long) change[3]);
        }
        verifiedChanges.clear();
        int moved = 0;
        for (Map.Entry<String, Map<String, Long>> branch : changes.entrySet()) {
            if (branch.getValue().isEmpty()) continue;
            boolean weekly = branch.getKey().endsWith("weekly");
            String place = branch.getKey() + (weekly ? "/" + thisWeek : "");
            Map<String, Object> counts = getMap("/counts/" + place + ".json");
            Map<String, Object> mirror = new LinkedHashMap<>();
            for (Map.Entry<String, Long> change : branch.getValue().entrySet()) {
                if (!userKey(change.getKey())) continue;
                Object before = Json.parse(orNull(call("GET", "/meta/counted/" + place + "/" + enc(change.getKey()) + ".json", null, false, null).body));
                long old = before instanceof Double ? ((Double) before).longValue() : 0L;
                long score = change.getValue();
                if (old == score) continue;
                if (old > 0) {
                    String key = SettingsActivity.Leaderboard.countKey((int) old);
                    long left = Math.max(0L, Json.number(counts, key)) - 1L;
                    if (left > 0) counts.put(key, (double) left);
                    else counts.remove(key);
                }
                String key = SettingsActivity.Leaderboard.countKey((int) score);
                counts.put(key, (double) (Math.max(0L, Json.number(counts, key)) + 1L));
                mirror.put(change.getKey(), (double) score);
                moved++;
            }
            if (mirror.isEmpty()) continue;
            counts.put(SettingsActivity.Leaderboard.COUNTS_DONE, (double) now);
            call("PUT", "/counts/" + place + ".json?print=silent", Json.write(counts), false, null);
            call("PATCH", "/meta/counted/" + place + ".json?print=silent", Json.write(mirror), false, null);
        }
        call("PATCH", "/meta/countedUpTo.json?print=silent", Json.write(reached), false, null);
        if (moved > 0) report.add("Ranks: " + moved + " new score" + (moved == 1 ? "" : "s") + " counted.");
    }

    static Map<String, Object> histogram(Map<String, Object> branch, long now) {
        Map<String, Object> counts = new java.util.TreeMap<>();
        for (Object raw : branch.values()) {
            long score = Json.number(Json.object(raw), "s");
            if (score <= 0 || score > SettingsActivity.Leaderboard.MAX_SCORE) continue;
            String key = SettingsActivity.Leaderboard.countKey((int) score);
            Object seen = counts.get(key);
            counts.put(key, (double) ((seen instanceof Double ? ((Double) seen).longValue() : 0L) + 1L));
        }
        counts.put(SettingsActivity.Leaderboard.COUNTS_DONE, (double) now);
        return counts;
    }

    static Map<String, Long> lastPlayed(List<String> users, long deadline) {
        Map<String, Long> out = new java.util.concurrent.ConcurrentHashMap<>();
        if (users.isEmpty()) return out;
        java.util.concurrent.ExecutorService pool = java.util.concurrent.Executors.newFixedThreadPool(8);
        try {
            List<java.util.concurrent.Future<?>> jobs = new ArrayList<>();
            for (String user : users) {
                jobs.add(pool.submit(() -> {
                    if (System.currentTimeMillis() > deadline) return;
                    try {
                        Object played = Json.parse(orNull(call("GET", "/sessions/" + enc(user) + "/t.json", null, false, null).body));
                        out.put(user, played instanceof Double ? ((Double) played).longValue() : 0L);
                    } catch (IOException | RuntimeException ignored) {
                    }
                }));
            }
            for (java.util.concurrent.Future<?> job : jobs) {
                try {
                    job.get();
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    break;
                } catch (java.util.concurrent.ExecutionException ignored) {
                }
            }
        } finally {
            pool.shutdownNow();
        }
        return out;
    }

    static void maintain(long now, long deadline, boolean force) throws IOException {
        Object stamp = Json.parse(orNull(call("GET", "/meta/maint.json", null, false, null).body));
        long last = stamp instanceof Double ? ((Double) stamp).longValue() : 0L;
        if (!force && last > 0 && now - last >= 0 && now - last < MAINTAIN_EVERY_MS) return;
        int thisWeek = SettingsActivity.Leaderboard.week(now);
        long weekFloor = thisWeek * SettingsActivity.Leaderboard.WEEK_KEY;
        Map<String, Object> update = new LinkedHashMap<>();
        int oldWeeks = 0, inactive = 0, names = 0, sessions = 0, verdicts = 0, seeds = 0;

        for (String branch : new String[] {"weekly", "vweekly"}) {
            Map<String, Object> old = getMap("/" + branch + ".json?orderBy=%22k%22&endAt=" + weekFloor);
            for (String user : old.keySet()) {
                remove(update, branch + "/" + user);
                oldWeeks++;
            }
        }
        flush(update);

        Map<String, Object> scores = getMap("/scores.json");
        Map<String, Object> vscores = getMap("/vscores.json");
        String range = "?orderBy=%22k%22&startAt=" + (weekFloor + 1) + "&endAt=" + (weekFloor + SettingsActivity.Leaderboard.MAX_SCORE);
        Map<String, Object> weekly = getMap("/weekly.json" + range);
        Map<String, Object> vweekly = getMap("/vweekly.json" + range);
        Object sweepStamp = Json.parse(orNull(call("GET", "/meta/sweepFrom.json", null, false, null).body));
        int lowered = 0;
        if (sweepStamp instanceof Double) {
            lowered = lowerStaleClaims(scores, vscores, weekly, vweekly, now, ((Double) sweepStamp).longValue(), update);
        } else {
            update.put("meta/sweepFrom", (double) now);
        }
        int lifted = expireBans(now, update);
        flush(update);
        java.util.Set<String> keep = new java.util.HashSet<>(top(scores, PROTECTED_TOP));
        keep.addAll(top(vscores, PROTECTED_TOP));
        keep.addAll(weekly.keySet());
        keep.addAll(vweekly.keySet());
        java.util.Set<String> players = new java.util.TreeSet<>(scores.keySet());
        players.addAll(vscores.keySet());
        Map<String, Object> claims = getMap("/names.json");
        Map<String, String> nameOf = new LinkedHashMap<>();
        for (Map.Entry<String, Object> claim : claims.entrySet()) {
            if (claim.getValue() instanceof String) nameOf.put((String) claim.getValue(), claim.getKey());
        }
        long cutoff = now - INACTIVE_MS;
        List<String> candidates = new ArrayList<>();
        for (String user : players) {
            if (keep.contains(user) || !userKey(user)) continue;
            long seen = Math.max(Json.number(Json.object(scores.get(user)), "t"), Json.number(Json.object(vscores.get(user)), "t"));
            if (seen < cutoff) candidates.add(user);
        }
        Map<String, Long> lastPlayed = lastPlayed(candidates, deadline);
        for (String user : candidates) {
            Long played = lastPlayed.get(user);
            if (played == null || played >= cutoff) continue;
            for (String branch : new String[] {"scores", "vscores", "weekly", "vweekly", "sessions", "verdicts", "replays"}) {
                remove(update, branch + "/" + user);
            }
            String key = nameOf.get(user);
            if (key != null && user.equals(claims.get(key))) {
                remove(update, "names/" + key);
                claims.remove(key);
            }
            scores.remove(user);
            vscores.remove(user);
            inactive++;
        }
        flush(update);

        Map<String, Object> orphans = getMap("/meta/orphans.json");
        Map<String, Object> stillOrphans = new LinkedHashMap<>();
        for (Map.Entry<String, Object> claim : claims.entrySet()) {
            String key = claim.getKey();
            Object owner = claim.getValue();
            if (owner instanceof String && (scores.containsKey(owner) || vscores.containsKey(owner))) continue;
            long first = Json.number(orphans, key);
            if (first <= 0) first = now;
            if (now - first >= INACTIVE_MS && System.currentTimeMillis() <= deadline) {
                Object played = owner instanceof String && userKey((String) owner)
                        ? Json.parse(orNull(call("GET", "/sessions/" + enc((String) owner) + "/t.json", null, false, null).body)) : null;
                if (!(played instanceof Double) || ((Double) played).longValue() < cutoff) {
                    remove(update, "names/" + key);
                    names++;
                    continue;
                }
                first = now;
            }
            stillOrphans.put(key, (double) first);
        }
        flush(update);

        Map<String, Object> staleSessions = getMap("/sessions.json?orderBy=%22t%22&endAt=" + cutoff);
        for (String user : staleSessions.keySet()) {
            remove(update, "sessions/" + user);
            sessions++;
        }
        Map<String, Object> oldVerdicts = getMap("/verdicts.json?orderBy=%22t%22&endAt=" + (now - VERDICT_KEEP_MS));
        for (String user : oldVerdicts.keySet()) {
            remove(update, "verdicts/" + user);
            verdicts++;
        }
        if (!Boolean.TRUE.equals(Json.parse(orNull(call("GET", "/meta/seedsTimed.json", null, false, null).body)))) {
            Map<String, Object> all = getMap("/seeds.json");
            for (Map.Entry<String, Object> seed : all.entrySet()) {
                if (seed.getValue() instanceof String) {
                    Map<String, Object> claim = new LinkedHashMap<>();
                    claim.put("u", seed.getValue());
                    claim.put("t", (double) now);
                    update.put("seeds/" + seed.getKey(), claim);
                    if (update.size() >= BATCH) flush(update);
                }
            }
            update.put("meta/seedsTimed", true);
        } else {
            Map<String, Object> oldSeeds = getMap("/seeds.json?orderBy=%22t%22&endAt=" + (now - SEED_KEEP_MS));
            for (String seed : oldSeeds.keySet()) {
                remove(update, "seeds/" + seed);
                seeds++;
            }
        }
        flush(update);

        Map<String, Object> counts = new LinkedHashMap<>();
        counts.put("scores", histogram(scores, now));
        counts.put("vscores", histogram(vscores, now));
        Map<String, Object> weeklyCounts = new LinkedHashMap<>();
        weeklyCounts.put(String.valueOf(thisWeek), histogram(weekly, now));
        counts.put("weekly", weeklyCounts);
        Map<String, Object> verifiedWeekCounts = new LinkedHashMap<>();
        verifiedWeekCounts.put(String.valueOf(thisWeek), histogram(vweekly, now));
        counts.put("vweekly", verifiedWeekCounts);
        call("PUT", "/counts.json?print=silent", Json.write(counts), false, null);
        Map<String, Object> mirror = new LinkedHashMap<>();
        mirror.put("scores", scoresOf(scores));
        mirror.put("vscores", scoresOf(vscores));
        Map<String, Object> weeklyMirror = new LinkedHashMap<>();
        weeklyMirror.put(String.valueOf(thisWeek), scoresOf(weekly));
        mirror.put("weekly", weeklyMirror);
        Map<String, Object> verifiedWeekMirror = new LinkedHashMap<>();
        verifiedWeekMirror.put(String.valueOf(thisWeek), scoresOf(vweekly));
        mirror.put("vweekly", verifiedWeekMirror);
        call("PUT", "/meta/counted.json?print=silent", Json.write(mirror), false, null);
        Map<String, Object> upTo = new LinkedHashMap<>();
        upTo.put("scores", (double) newest(scores));
        upTo.put("weekly", (double) newest(weekly));
        call("PUT", "/meta/countedUpTo.json?print=silent", Json.write(upTo), false, null);
        verifiedChanges.clear();

        Map<String, Object> done = new LinkedHashMap<>();
        done.put("meta/maint", (double) now);
        done.put("meta/orphans", stillOrphans.isEmpty() ? null : stillOrphans);
        call("PATCH", "/.json?print=silent", Json.write(done), false, null);
        report.add("Cleanup: removed " + oldWeeks + " old weekly entries, " + inactive + " inactive players, " + names + " unused names, "
                + sessions + " old rounds, " + verdicts + " old results and " + seeds + " old seeds. Ranks updated for "
                + scores.size() + " players.");
        if (lowered > 0) report.add("Took down " + lowered + " unverified score" + (lowered == 1 ? "" : "s") + " that had no run sent within a day.");
        if (lifted > 0) report.add("Lifted " + lifted + " automatic ban" + (lifted == 1 ? "" : "s") + " after " + (AUTO_BAN_MS / DAY_MS) + " days.");
    }

    static String orNull(String body) {
        String trimmed = body == null ? "" : body.trim();
        return trimmed.isEmpty() ? "null" : trimmed;
    }

    static void publish(Map<String, Object> update, String user, String name, int score, boolean classic, long sent, int week,
            Replayer.Result result, Map<String, Object> scores, String version) throws IOException {
        double seconds = Math.ceil(result.seconds);
        Map<String, Object> style = styleOf(user);
        Map<String, Object> best = Json.object(Json.parse(call("GET", "/vscores/" + enc(user) + ".json", null, false, null).body));
        if (best == null || Json.number(best, "s") < score) {
            Map<String, Object> entry = new LinkedHashMap<>();
            if (style != null) entry.put("y", style);
            entry.put("n", name);
            entry.put("s", (double) score);
            entry.put("c", classic);
            entry.put("t", (double) sent);
            entry.put("d", seconds);
            if (!version.isEmpty()) entry.put("vn", version);
            update.put("vscores/" + user, entry);
            verifiedChanges.add(new Object[] {"vscores", -1, user, (long) score});
        }
        if (week == SettingsActivity.Leaderboard.week(sent)) {
            Map<String, Object> weekly = Json.object(Json.parse(call("GET", "/vweekly/" + enc(user) + ".json", null, false, null).body));
            long savedWeek = Json.number(weekly, "w");
            if (weekly == null || savedWeek < week || (savedWeek == week && Json.number(weekly, "s") < score)) {
                Map<String, Object> entry = new LinkedHashMap<>();
                if (style != null) entry.put("y", style);
                entry.put("n", name);
                entry.put("s", (double) score);
                entry.put("c", classic);
                entry.put("t", (double) sent);
                entry.put("d", seconds);
                entry.put("w", (double) week);
                entry.put("k", (double) (week * SettingsActivity.Leaderboard.WEEK_KEY + score));
                if (!version.isEmpty()) entry.put("vn", version);
                update.put("vweekly/" + user, entry);
                verifiedChanges.add(new Object[] {"vweekly", week, user, (long) score});
            }
        }
        if (Json.number(scores, "s") == score) update.put("scores/" + user + "/vs", (double) score);
        Map<String, Object> posted = Json.object(Json.parse(call("GET", "/weekly/" + enc(user) + ".json", null, false, null).body));
        if (Json.number(posted, "k") == week * SettingsActivity.Leaderboard.WEEK_KEY + score) update.put("weekly/" + user + "/vs", (double) score);
    }

    static Map<String, Object> styleOf(String user) throws IOException {
        Map<String, Object> wallet = getMap("/wallet/" + enc(user) + ".json");
        Map<String, Object> style = Json.object(wallet.get("y"));
        Map<String, Object> owned = Json.object(wallet.get("o"));
        if (style == null || owned == null) return null;
        Map<String, Object> kept = new LinkedHashMap<>();
        for (Map.Entry<String, Object> part : style.entrySet()) {
            String kind = part.getKey();
            Object id = part.getValue();
            if (kind.length() == 1 && Shop.KINDS.contains(kind) && id instanceof String && Shop.validId((String) id)
                    && ((String) id).startsWith(kind + "_") && owned.containsKey(id)) {
                kept.put(kind, id);
            }
        }
        return kept.isEmpty() ? null : kept;
    }

    static int weekBest(String user, long sent) throws IOException {
        int week = SettingsActivity.Leaderboard.week(sent);
        long best = 0L;
        Map<String, Object> weekly = getMap("/vweekly/" + enc(user) + ".json");
        if (Json.number(weekly, "w") == week) best = Math.max(best, Json.number(weekly, "s"));
        Map<String, Object> kept = getMap("/wallet/" + enc(user) + "/k.json");
        if (Json.number(kept, "w") == week) best = Math.max(best, Json.number(kept, "s"));
        return (int) Math.max(0L, Math.min(SettingsActivity.Leaderboard.MAX_SCORE, best));
    }

    static void news(Map<String, Object> update, String user, int gain, int newly, boolean ok, String why) {
        Map<String, Object> note = new LinkedHashMap<>();
        Map<String, Object> stamp = new LinkedHashMap<>();
        stamp.put(".sv", "timestamp");
        note.put("t", stamp);
        note.put("g", (double) gain);
        note.put("m", (double) newly);
        note.put("ok", ok);
        if (why != null && !why.isEmpty()) note.put("why", why.length() > 160 ? why.substring(0, 160) : why);
        update.put("wallet/" + user + "/n", note);
    }

    static void awardFromSeed(String user, String seed, long sent, int score) throws IOException {
        Map<String, Object> record = getMap("/seeds/" + enc(seed) + ".json");
        Map<String, Object> facts = Json.object(record.get("f"));
        if (facts == null || Boolean.TRUE.equals(record.get("q")) || Json.number(facts, "s") != score) return;
        Map<String, Object> update = new LinkedHashMap<>();
        int restMode = (int) Json.number(facts, "md");
        Object seconds = facts.get("sec");
        int gained = awardRound(update, user, sent, score, seconds instanceof Number ? ((Number) seconds).doubleValue() : 0.0,
                (int) Json.number(facts, "fl"), Challenges.ANY, restMode, (int) Math.max(0L, Json.number(facts, "wb")));
        update.put("seeds/" + seed + "/q", true);
        call("PATCH", "/.json", Json.write(update), false, null);
        report.add("Counted a leaderboard run of " + score + " by " + shortId(user) + " toward challenges" + (gained > 0 ? ": earned " + gained + " coins" : ""));
    }

    static int awardRound(Map<String, Object> update, String user, long sent, int score, double seconds, int flaps, int twist, int restMode,
            int weekBest) throws IOException {
        Map<String, Object> wallet = getMap("/wallet/" + enc(user) + ".json");
        long day = Challenges.day(sent);
        Map<String, Object> daily = Json.object(wallet.get("d"));
        long savedDay = daily == null ? Long.MIN_VALUE : Json.number(daily, "d");
        if (savedDay != Long.MIN_VALUE && savedDay > day) {
            news(update, user, 0, 0, false, "the run was from an earlier day");
            return 0;
        }
        Challenges.Goal[] goals = Challenges.forDay(day);
        Map<String, Object> next = new LinkedHashMap<>();
        next.put("d", (double) day);
        int mask = 0;
        int[] progress = new int[3];
        if (daily != null && savedDay == day) {
            String ids = Json.string(daily, "g");
            String[] parts = ids.split(",");
            if (parts.length == 3) {
                for (int i = 0; i < 3; i++) {
                    Challenges.Goal goal = Challenges.find(parts[i]);
                    goals[i] = goal != null && goal.tier == i ? goal : null;
                }
            }
            mask = (int) (Json.number(daily, "m") & 7L);
            if (mask < 0) mask = 0;
            for (int i = 0; i < 3; i++) progress[i] = (int) Math.max(0L, Json.number(daily, "p" + i));
        }
        StringBuilder ids = new StringBuilder();
        for (int i = 0; i < 3; i++) {
            if (i > 0) ids.append(',');
            ids.append(goals[i] == null ? "" : goals[i].id);
        }
        next.put("g", ids.toString());
        int gain = 0;
        int newly = 0;
        for (int i = 0; i < 3; i++) {
            Challenges.Goal goal = goals[i];
            if (goal == null || (mask & (1 << i)) != 0 || !Challenges.fits(goal, twist, restMode)) continue;
            progress[i] = Challenges.advance(goal, progress[i], score, seconds, flaps, weekBest);
            if (Challenges.done(goal, progress[i])) {
                mask |= 1 << i;
                newly |= 1 << i;
                gain += Challenges.REWARD[i];
            }
        }
        for (int i = 0; i < 3; i++) next.put("p" + i, (double) progress[i]);
        next.put("m", (double) mask);
        update.put("wallet/" + user + "/d", next);
        if (mask == 7 && newly != 0) {
            long last = wallet.containsKey("l") ? Json.number(wallet, "l") : Long.MIN_VALUE;
            long streak = Math.max(0L, Json.number(wallet, "s"));
            streak = last == day - 1 ? streak + 1 : last == day ? Math.max(1L, streak) : 1L;
            update.put("wallet/" + user + "/s", (double) streak);
            update.put("wallet/" + user + "/l", (double) day);
            gain += Challenges.bonus((int) Math.min(100000L, streak));
        }
        if (twist == Challenges.ANY && restMode != SettingsActivity.FAIR_NONE) {
            int week = SettingsActivity.Leaderboard.week(sent);
            Map<String, Object> kept = Json.object(wallet.get("k"));
            long best = kept != null && Json.number(kept, "w") == week ? Json.number(kept, "s") : 0L;
            if (score > best) {
                Map<String, Object> mark = new LinkedHashMap<>();
                mark.put("w", (double) week);
                mark.put("s", (double) score);
                update.put("wallet/" + user + "/k", mark);
            }
        }
        if (gain > 0) {
            long balance = Math.max(0L, Json.number(wallet, "b"));
            long earned = Math.max(0L, Json.number(wallet, "e"));
            update.put("wallet/" + user + "/b", (double) Math.min(Challenges.MAX_BALANCE, balance + gain));
            update.put("wallet/" + user + "/e", (double) Math.min(Integer.MAX_VALUE, earned + gain));
        }
        news(update, user, gain, newly, true, "");
        return gain;
    }

    static void processPurchases(long deadline) throws IOException {
        Map<String, Object> waitingBuys = getMap("/buy.json?shallow=true");
        for (String user : waitingBuys.keySet()) {
            if (System.currentTimeMillis() > deadline) return;
            if (!userKey(user)) {
                call("DELETE", "/buy/" + enc(user) + ".json", null, false, null);
                continue;
            }
            Reply reply = call("GET", "/buy/" + enc(user) + ".json", null, true, null);
            Map<String, Object> order = Json.object(Json.parse(orNull(reply.body)));
            if (order == null) continue;
            String id = Json.string(order, "i");
            Shop.Item item = Shop.item(id);
            Map<String, Object> wallet = getMap("/wallet/" + enc(user) + ".json");
            Map<String, Object> owned = Json.object(wallet.get("o"));
            long balance = Math.max(0L, Json.number(wallet, "b"));
            Map<String, Object> update = new LinkedHashMap<>();
            boolean ok;
            String result;
            if (item == null) {
                ok = false;
                result = "an unknown item";
            } else if (exists("/banned/" + enc(user) + ".json")) {
                ok = false;
                result = "a banned player";
            } else if (owned != null && owned.containsKey(item.id)) {
                ok = true;
                result = "already owned";
            } else if (balance >= item.price) {
                ok = true;
                result = "bought for " + item.price;
                update.put("wallet/" + user + "/b", (double) (balance - item.price));
                update.put("wallet/" + user + "/o/" + item.id, true);
            } else {
                ok = false;
                result = "not enough coins (" + balance + " of " + item.price + ")";
            }
            Map<String, Object> done = new LinkedHashMap<>();
            done.put("i", item == null ? "" : item.id);
            done.put("ok", ok);
            Map<String, Object> stamp = new LinkedHashMap<>();
            stamp.put(".sv", "timestamp");
            done.put("t", stamp);
            update.put("wallet/" + user + "/r", done);
            call("PATCH", "/.json", Json.write(update), false, null);
            Reply removed = call("DELETE", "/buy/" + enc(user) + ".json", null, false, reply.etag);
            if (removed.code == 412) report.add("A purchase from " + shortId(user) + " changed while it was handled; it will be looked at again.");
            report.add("Shop: " + shortId(user) + " asked for " + (item == null ? "something unknown" : item.name) + ", " + result + ".");
        }
    }

    static boolean exists(String path) throws IOException {
        String body = call("GET", path + (path.contains("?") ? "&" : "?") + "shallow=true", null, false, null).body.trim();
        return !body.isEmpty() && !body.equals("null");
    }

    static Reply call(String method, String path, String body, boolean wantTag, String ifMatch) throws IOException {
        String address = database + path + (path.contains("?") ? "&" : "?") + "access_token=" + enc(accessToken);
        HttpURLConnection connection = (HttpURLConnection) new URL(address).openConnection();
        try {
            boolean patch = "PATCH".equals(method);
            connection.setRequestMethod(patch ? "POST" : method);
            if (patch) connection.setRequestProperty("X-HTTP-Method-Override", "PATCH");
            connection.setConnectTimeout(TIMEOUT_MS);
            connection.setReadTimeout(TIMEOUT_MS);
            connection.setUseCaches(false);
            connection.setRequestProperty("Accept", "application/json");
            if (wantTag) connection.setRequestProperty("X-Firebase-ETag", "true");
            if (ifMatch != null) connection.setRequestProperty("if-match", ifMatch);
            if (body != null) {
                connection.setDoOutput(true);
                connection.setRequestProperty("Content-Type", "application/json; charset=utf-8");
                try (OutputStream out = connection.getOutputStream()) {
                    out.write(body.getBytes(StandardCharsets.UTF_8));
                }
            }
            Reply reply = new Reply();
            reply.code = connection.getResponseCode();
            reply.etag = connection.getHeaderField("ETag");
            InputStream stream = reply.code >= 400 ? connection.getErrorStream() : connection.getInputStream();
            reply.body = stream == null ? "" : read(stream);
            if (reply.code >= 400 && reply.code != 412) {
                throw new IOException(method + " " + path.replaceAll("\\?.*", "") + " failed with " + reply.code + ": "
                        + (reply.body.length() > 200 ? reply.body.substring(0, 200) : reply.body));
            }
            return reply;
        } finally {
            connection.disconnect();
        }
    }

    static String read(InputStream stream) throws IOException {
        try (InputStream in = stream) {
            ByteArrayOutputStream bytes = new ByteArrayOutputStream();
            byte[] buffer = new byte[16384];
            int count;
            while ((count = in.read(buffer)) != -1) bytes.write(buffer, 0, count);
            return new String(bytes.toByteArray(), StandardCharsets.UTF_8);
        }
    }

    static String base64Url(byte[] bytes) {
        return Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);
    }

    static String signIn(Map<String, Object> key) throws Exception {
        String email = Json.string(key, "client_email");
        String tokenUri = Json.string(key, "token_uri");
        if (tokenUri.isEmpty()) tokenUri = "https://oauth2.googleapis.com/token";
        String pem = Json.string(key, "private_key").replace("-----BEGIN PRIVATE KEY-----", "").replace("-----END PRIVATE KEY-----", "")
                .replaceAll("\\s", "");
        PrivateKey privateKey = KeyFactory.getInstance("RSA").generatePrivate(new PKCS8EncodedKeySpec(Base64.getDecoder().decode(pem)));
        long now = System.currentTimeMillis() / 1000L;
        Map<String, Object> claims = new LinkedHashMap<>();
        claims.put("iss", email);
        claims.put("scope", SCOPES);
        claims.put("aud", tokenUri);
        claims.put("iat", (double) now);
        claims.put("exp", (double) (now + 3600));
        String unsigned = base64Url("{\"alg\":\"RS256\",\"typ\":\"JWT\"}".getBytes(StandardCharsets.UTF_8)) + "."
                + base64Url(Json.write(claims).getBytes(StandardCharsets.UTF_8));
        Signature signer = Signature.getInstance("SHA256withRSA");
        signer.initSign(privateKey);
        signer.update(unsigned.getBytes(StandardCharsets.UTF_8));
        String jwt = unsigned + "." + base64Url(signer.sign());
        String form = "grant_type=" + enc("urn:ietf:params:oauth:grant-type:jwt-bearer") + "&assertion=" + enc(jwt);
        HttpURLConnection connection = (HttpURLConnection) new URL(tokenUri).openConnection();
        try {
            connection.setRequestMethod("POST");
            connection.setConnectTimeout(TIMEOUT_MS);
            connection.setReadTimeout(TIMEOUT_MS);
            connection.setDoOutput(true);
            connection.setRequestProperty("Content-Type", "application/x-www-form-urlencoded");
            try (OutputStream out = connection.getOutputStream()) {
                out.write(form.getBytes(StandardCharsets.UTF_8));
            }
            int code = connection.getResponseCode();
            InputStream stream = code >= 400 ? connection.getErrorStream() : connection.getInputStream();
            String body = stream == null ? "" : read(stream);
            String token = Json.string(Json.object(Json.parse(body)), "access_token");
            if (code >= 400 || token.isEmpty()) throw new IOException("Google refused the service account key (" + code + ")");
            return token;
        } finally {
            connection.disconnect();
        }
    }
