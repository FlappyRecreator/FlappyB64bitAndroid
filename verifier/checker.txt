    private Bitmap(int w, int h, boolean mutable) {

    }

    public static Bitmap createBitmap(Bitmap src, int x, int y, int w, int h, Matrix m, boolean filter) {
        if (src.recycled) throw new IllegalStateException("createBitmap from recycled bitmap");
        if (x < 0 || y < 0) throw new IllegalArgumentException("x/y must be >= 0");
        if (x + w > src.w) throw new IllegalArgumentException("x + width must be <= bitmap.width()");
        if (y + h > src.h) throw new IllegalArgumentException("y + height must be <= bitmap.height()");
        if (!src.mutable && x == 0 && y == 0 && w == src.w && h == src.h && (m == null || m.isIdentity())) return src;
        int nw = w, nh = h;
        if (m != null && !m.isIdentity()) { nw = Math.round(Math.abs(w * m.sx)); nh = Math.round(Math.abs(h * m.sy)); }
        return new Bitmap(nw, nh, false);
    }
    public static Bitmap createScaledBitmap(Bitmap src, int dw, int dh, boolean filter) {
        Matrix m = new Matrix();
        if (src.w != dw || src.h != dh) m.setScale(dw / (float) src.w, dh / (float) src.h);
        if (dw <= 0 || dh <= 0) throw new IllegalArgumentException("width and height must be > 0");
        Bitmap b = createBitmap(src, 0, 0, src.w, src.h, m, filter);
        if (b != src && (b.w != dw || b.h != dh)) return new Bitmap(dw, dh, false);
        return b;
    }

    private static Map<String, Object> stamp(Object stamps, int index) {

        if (stamps instanceof Map) one = ((Map<String, Object>) stamps).get(String.valueOf(index));

    @SuppressWarnings("unchecked")
    private static int stampCount(Object stamps) {
        if (stamps instanceof Map) return ((Map<String, Object>) stamps).size();

    private static long whole(Object value) {
        if (!(value instanceof Number)) return Long.MIN_VALUE;
        double d = ((Number) value).doubleValue();
        return Double.isNaN(d) || Double.isInfinite(d) || d != Math.rint(d) ? Long.MIN_VALUE : (long) d;
    }

    static String checkTiming(Result result, Object stamps, long started, long uploaded) {
        List<Mark> marks = result.marks;
        if (marks.isEmpty() || stampCount(stamps) != marks.size()) return "the run's timing marks didn't match the replay";
        long previous = started;
        long pausedMs = 0L;
        long pausedAt = -1L;
        boolean playing = true;
        double slowest = 0.0;
        for (int i = 0; i < marks.size(); i++) {
            Mark mark = marks.get(i);
            Map<String, Object> one = stamp(stamps, i);
            if (one == null) return "the run's timing marks didn't match the replay";
            long time = whole(one.get("t"));
            if (whole(one.get("m")) != mark.kind || !mark.code.equals(one.get("c")) || time == Long.MIN_VALUE) {
                return "the run's timing marks didn't match the replay";
            }
            if (time < previous && i > 0) return "the run's timing marks were out of order";
            if (playing && time - previous > MAX_GAP_MS) {
                return "the run's timing could not be confirmed (no mark for " + (time - previous) / 1000 + " s)";
            }
            if (mark.kind == GameView.MARK_PAUSE) {
                pausedAt = time;
                playing = false;
            } else if (mark.kind == GameView.MARK_RESUME) {
                pausedMs += time - pausedAt;
                playing = true;
            }
            double real = (time - started - pausedMs) / 1000.0;
            if (mark.seconds > real + FAST_SLACK_SEC) return "the run was played faster than real time";
            double lag = real - mark.seconds * SLOW_SHARE;
            if (lag > slowest + SLOW_SLACK_SEC) return "the run was played in slow motion";
            slowest = Math.min(slowest, lag);
            previous = time;
        }
        if (marks.get(marks.size() - 1).kind != GameView.MARK_END) return "the replay has no end mark";
        if (uploaded < previous) return "the run's timing marks were out of order";
        return null;
    }
